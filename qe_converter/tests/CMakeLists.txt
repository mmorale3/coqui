# Phase 0 schema validator for pw2coqui output.
#
# The converter itself lives in QE's build tree (see qe_converter/README.md);
# we don't build pw2coqui.x from CoQui's CMake. We just register a CTest that
# runs check_schema.sh against any *.coqui.h5 fixture in
# tests/unit_test_files/qe/. The test SKIPs (exit 77) if h5dump is missing.

find_program(H5DUMP_EXE h5dump)

if(NOT H5DUMP_EXE)
    message(STATUS "qe_converter tests: h5dump not found, schema test will skip")
endif()

set(_validator "${CMAKE_CURRENT_SOURCE_DIR}/check_schema.sh")
set(_fixture_root
    "${PROJECT_SOURCE_DIR}/tests/unit_test_files/qe")

# Collect fixture .h5 files that exist on disk at configure time and that
# already carry the Phase 0 schema (have a /Hamiltonian/Species group).
# Fixtures generated by the pre-Phase-0 converter are skipped here so that
# the test suite isn't pre-broken on first-time check-out; regenerate the
# fixture with the new pw2coqui.x to opt it in.
file(GLOB _fixtures "${_fixture_root}/*/pwscf.coqui.h5")

set(_filtered_fixtures "")
if(H5DUMP_EXE)
    foreach(_h5 IN LISTS _fixtures)
        execute_process(
            COMMAND "${H5DUMP_EXE}" -n "${_h5}"
            OUTPUT_VARIABLE _h5_listing
            RESULT_VARIABLE _h5_rc
            ERROR_QUIET
        )
        if(_h5_rc EQUAL 0 AND _h5_listing MATCHES "/Hamiltonian/Species")
            list(APPEND _filtered_fixtures "${_h5}")
        endif()
    endforeach()
else()
    set(_filtered_fixtures ${_fixtures})
endif()

foreach(_h5 IN LISTS _filtered_fixtures)
    get_filename_component(_dir "${_h5}" DIRECTORY)
    get_filename_component(_name "${_dir}" NAME)
    set(_test_name "qe_converter_schema_${_name}")
    add_test(
        NAME ${_test_name}
        COMMAND bash "${_validator}" "${_h5}"
    )
    set_tests_properties(${_test_name} PROPERTIES
        LABELS "schema;qe_converter"
        SKIP_RETURN_CODE 77
    )
endforeach()

list(LENGTH _filtered_fixtures _n_fixtures)
if(_n_fixtures EQUAL 0)
    message(STATUS
        "qe_converter tests: no Phase 0 fixtures under ${_fixture_root} "
        "(regenerate with the new pw2coqui.x to enable schema tests)")
else()
    message(STATUS
        "qe_converter tests: registered ${_n_fixtures} schema test(s)")
endif()
